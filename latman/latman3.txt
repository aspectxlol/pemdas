Menghitung Pecahan Uang

DEKLARASI
    masuk : integer
    pecahan_uang : array of string
    jumlah_pecahan_seratus : integer
    jumlah_pecahan_lima_puluh_ribu : integer
    jumlah_pecahan_dua_puluh_ribu : integer
    jumlah_pecahan_sepuluh_ribu : integer
    jumlah_pecahan_lima_ribu : integer
    jumlah_pecahan_dua_ribu : integer
    jumlah_pecahan_seribu : integer
    jumlah_sisa : integer

PROGRAM UTAMA
    pecahan_uang ← ["seribu", "dua ribu", "lima ribu",
                    "sepuluh ribu", "dua puluh ribu",
                    "lima puluh ribu", "seratus ribu"]

    INPUT masuk

    jumlah_pecahan_seratus ← masuk // 100000
    jumlah_pecahan_lima_puluh_ribu ← (masuk % 100000) // 50000
    jumlah_pecahan_dua_puluh_ribu ← (masuk % 50000) // 20000
    jumlah_pecahan_sepuluh_ribu ← (masuk % 20000) // 10000
    jumlah_pecahan_lima_ribu ← (masuk % 10000) // 5000
    jumlah_pecahan_dua_ribu ← (masuk % 5000) // 2000
    jumlah_pecahan_seribu ← (masuk % 2000) // 1000
    jumlah_sisa ← masuk % 1000

    OUTPUT "Pecahan uang:"
    OUTPUT jumlah_pecahan_seratus, " = uang ", pecahan_uang[6]
    OUTPUT jumlah_pecahan_lima_puluh_ribu, " = uang ", pecahan_uang[5]
    OUTPUT jumlah_pecahan_dua_puluh_ribu, " = uang ", pecahan_uang[4]
    OUTPUT jumlah_pecahan_sepuluh_ribu, " = uang ", pecahan_uang[3]
    OUTPUT jumlah_pecahan_lima_ribu, " = uang ", pecahan_uang[2]
    OUTPUT jumlah_pecahan_dua_ribu, " = uang ", pecahan_uang[1]
    OUTPUT jumlah_pecahan_seribu, " = uang ", pecahan_uang[0]
    OUTPUT "Tersisa: ", jumlah_sisa, " rupiah"
